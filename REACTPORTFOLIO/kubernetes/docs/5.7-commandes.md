# 5.7 Commandes de base Kubernetes

## Informations du cluster

```bash
# Info du cluster
kubectl cluster-info

# Lister les noeuds
kubectl get nodes
kubectl get nodes -o wide

# Version
kubectl version
```

## Namespaces

```bash
# Lister les namespaces
kubectl get namespaces

# Creer un namespace
kubectl create namespace portfolio

# Definir le namespace par defaut
kubectl config set-context --current --namespace=portfolio
```

## Pods

```bash
# Lister les pods
kubectl get pods
kubectl get pods -n portfolio
kubectl get pods -o wide                    # avec IP et noeud
kubectl get pods --all-namespaces           # tous les namespaces

# Details d'un pod
kubectl describe pod <pod-name> -n portfolio

# Logs
kubectl logs <pod-name> -n portfolio
kubectl logs -f <pod-name> -n portfolio     # follow en temps reel
kubectl logs <pod-name> -c <container>      # multi-conteneur

# Executer une commande dans un pod
kubectl exec -it <pod-name> -n portfolio -- sh
kubectl exec -it <pod-name> -n portfolio -- node -v

# Supprimer un pod
kubectl delete pod <pod-name> -n portfolio
```

## Deployments

```bash
# Lister
kubectl get deployments -n portfolio

# Creer depuis un fichier
kubectl apply -f backend-deployment.yaml

# Scaler
kubectl scale deployment backend --replicas=4 -n portfolio

# Mettre a jour l'image
kubectl set image deployment/backend backend=portfolio-backend:v2 -n portfolio

# Voir l'historique des rollouts
kubectl rollout history deployment/backend -n portfolio

# Statut du rollout
kubectl rollout status deployment/backend -n portfolio

# Rollback
kubectl rollout undo deployment/backend -n portfolio
kubectl rollout undo deployment/backend --to-revision=2 -n portfolio

# Redemarrer tous les pods
kubectl rollout restart deployment/backend -n portfolio
```

## Services

```bash
# Lister
kubectl get services -n portfolio
kubectl get svc -n portfolio

# Details
kubectl describe svc backend-service -n portfolio

# Exposer un deployment
kubectl expose deployment backend --port=5000 --type=ClusterIP -n portfolio

# Port-forward (acces local temporaire)
kubectl port-forward svc/backend-service 5000:5000 -n portfolio
kubectl port-forward svc/frontend-service 3000:80 -n portfolio
```

## ConfigMaps et Secrets

```bash
# Lister
kubectl get configmaps -n portfolio
kubectl get secrets -n portfolio

# Voir le contenu
kubectl describe configmap portfolio-config -n portfolio
kubectl get secret portfolio-secret -n portfolio -o yaml

# Decoder un secret
kubectl get secret portfolio-secret -n portfolio -o jsonpath='{.data.MONGO_URI}' | base64 -d

# Creer un secret depuis la CLI
kubectl create secret generic my-secret --from-literal=password=mypass -n portfolio
```

## Volumes

```bash
# PersistentVolumes
kubectl get pv
kubectl describe pv mongo-pv

# PersistentVolumeClaims
kubectl get pvc -n portfolio
kubectl describe pvc mongo-pvc -n portfolio
```

## Ingress

```bash
# Lister
kubectl get ingress -n portfolio

# Details
kubectl describe ingress portfolio-ingress -n portfolio
```

## HPA (Horizontal Pod Autoscaler)

```bash
# Lister
kubectl get hpa -n portfolio

# Details
kubectl describe hpa backend-hpa -n portfolio

# Creer depuis la CLI
kubectl autoscale deployment backend --min=2 --max=5 --cpu-percent=70 -n portfolio
```

## StatefulSets

```bash
# Lister
kubectl get statefulsets -n portfolio

# Scaler
kubectl scale statefulset mongo --replicas=3 -n portfolio
```

## Debug et diagnostic

```bash
# Evenements du namespace
kubectl get events -n portfolio --sort-by='.lastTimestamp'

# Top (utilisation ressources)
kubectl top nodes
kubectl top pods -n portfolio

# Verifier la connectivite
kubectl run debug --image=busybox -it --rm -n portfolio -- wget -qO- http://backend-service:5000/api/projects

# DNS debug
kubectl run dns-debug --image=busybox -it --rm -n portfolio -- nslookup backend-service.portfolio.svc.cluster.local
```

## Gestion declarative (fichiers YAML)

```bash
# Appliquer un fichier
kubectl apply -f deployment.yaml

# Appliquer un repertoire entier
kubectl apply -f kubernetes/

# Supprimer depuis un fichier
kubectl delete -f deployment.yaml

# Voir les differences avant d'appliquer
kubectl diff -f deployment.yaml

# Generer un YAML sans creer
kubectl create deployment test --image=nginx --dry-run=client -o yaml
```

## Commandes utiles pour le projet

```bash
# Deployer tout le projet
cd kubernetes && kubectl apply -f .

# Voir l'etat complet
kubectl get all -n portfolio

# Suivre les logs du backend en temps reel
kubectl logs -f -l app=backend -n portfolio

# Tester le backend depuis le cluster
kubectl run curl --image=curlimages/curl -it --rm -n portfolio -- curl http://backend-service:5000/api/projects

# Forcer le redemarrage apres une mise a jour d'image
kubectl rollout restart deployment/frontend deployment/backend -n portfolio
```
